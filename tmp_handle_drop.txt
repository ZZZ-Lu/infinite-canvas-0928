  const handleDrop = useCallback(async (e: React.DragEvent) => {
    e.preventDefault();
    e.stopPropagation();
    setIsDragOver(false);

    if (!e.dataTransfer || !e.dataTransfer.files) return;
    const files = Array.from(e.dataTransfer.files) as File[];
    if (files.length === 0) return;

    const rect = containerRef.current?.getBoundingClientRect();
    if (!rect) return;

    // Calculate canvas drop coordinates
    const dropClientX = e.clientX;
    const dropClientY = e.clientY;
    const dropCanvasX = (dropClientX - tx.get()) / tScale.get();
    const dropCanvasY = (dropClientY - ty.get()) / tScale.get();

    // Step 1: Fast initialization for immediate visual feedback (Zero blocking)
    const initialCards: CardData[] = [];
    const validFiles: { 
      file: File, isImage: boolean, isVideo: boolean, id: string, 
      offset: number, fileUrl: string
    }[] = [];

    for (let i = 0; i < files.length; i++) {
      const file = files[i];
      const isImage = file.type.startsWith('image/');
      const isVideo = file.type.startsWith('video/');
      if (!isImage && !isVideo) continue;

      const fileUrl = URL.createObjectURL(file);
      const newId = Math.random().toString(36).substring(2, 11);
      const offset = i * 40;
      const defaultDim = CARD_DIMENSIONS['1:1'];

      initialCards.push({
        id: newId,
        x: dropCanvasX - defaultDim.width / 2 + offset,
        y: dropCanvasY - defaultDim.height / 2 + offset,
        state: 'generating',
        ratio: '1:1', // Temporary ratio
        res: '2K',
        prompt: `Dropped local ${isVideo ? 'video' : 'image'}: ${file.name}`,
        imageUrl: null, // Generating spinner will show
        isVideo: isVideo,
      });

      validFiles.push({ file, isImage, isVideo, id: newId, offset, fileUrl });
    }

    if (initialCards.length === 0) return;

    // Step 2: Show placeholders on canvas immediately
    setCards(prev => [...prev, ...initialCards]);
    setSelectedCardIds(initialCards.map(c => c.id));

    // Step 3: Start heavy background processing asynchronously
    validFiles.forEach(async (item) => {
      // Read dimensions
      let width = 0;
      let height = 0;
      try {
        if (item.isImage) {
          const dims = await getImageDimensions(item.fileUrl);
          width = dims.width;
          height = dims.height;
        } else if (item.isVideo) {
          const dims = await getVideoDimensions(item.fileUrl);
          width = dims.width;
          height = dims.height;
        }
      } catch (e) {
        console.error("Failed to read dimensions", e);
      }

      const ratio = getClosestAspectRatio(width, height);
      let customWidth: number | undefined;
      let customHeight: number | undefined;
      if (width > 0 && height > 0) {
        const TARGET_AREA = 230400; // 480x480 standard equivalent
        const scaleFactor = Math.sqrt(TARGET_AREA / (width * height));
        customWidth = Math.round(width * scaleFactor);
        customHeight = Math.round(height * scaleFactor);
      }

      // Re-center based on actual dimensions
      const finalDim = customWidth && customHeight ? { width: customWidth, height: customHeight } : CARD_DIMENSIONS[ratio];
      const finalX = dropCanvasX - finalDim.width / 2 + item.offset;
      const finalY = dropCanvasY - finalDim.height / 2 + item.offset;

      let processedFile: Blob = item.file;
      let originalFileData: Blob | undefined = undefined;
      let trueOriginalFileData: Blob | undefined = undefined;
      let originalImageUrl: string | undefined = undefined;
      let trueOriginalImageUrl: string | undefined = undefined;

      if (item.isImage) {
        try {
          // Generate 1200px preview
          processedFile = await compressAndResizeImage(item.file, 1200);
          
          // Check if original is > 10000px (GPU texture limit safety)
          const MAX_TEXTURE_DIM = 10000;
          if (width > MAX_TEXTURE_DIM || height > MAX_TEXTURE_DIM) {
            trueOriginalFileData = item.file;
            trueOriginalImageUrl = URL.createObjectURL(item.file);
            
            // Create a safe proxy for the "original" view in the UI to prevent GPU crashes
            originalFileData = await compressAndResizeImage(item.file, MAX_TEXTURE_DIM, 0.9);
            originalImageUrl = URL.createObjectURL(originalFileData);
          } else {
            // It's under 10000px, so the original file is the proxy itself
            originalFileData = item.file;
            originalImageUrl = URL.createObjectURL(item.file);
          }
        } catch (err) {
          console.error("Failed to process image", err);
          processedFile = await compressAndResizeImage(item.file, 1200);
          originalFileData = item.file;
          originalImageUrl = URL.createObjectURL(item.file);
        }
      }

      const processedFileUrl = URL.createObjectURL(processedFile);

      let thumbnailUrl: string | undefined;
      try {
        if (item.isVideo) {
          thumbnailUrl = await generateVideoThumbnail(processedFile, MAX_THUMBNAIL_EDGE);
        } else {
          thumbnailUrl = await generateImageThumbnail(processedFile, MAX_THUMBNAIL_EDGE);
        }
      } catch (thumbErr) {
        console.warn('Could not pre-generate thumbnail on drop:', thumbErr);
      }

      // Step 4: Update the specific card with completed data and true dimensions
      setCards(prev => prev.map(c => c.id === item.id ? {
        ...c,
        x: finalX,
        y: finalY,
        state: 'completed',
        ratio: ratio,
        customWidth: customWidth,
        customHeight: customHeight,
        imageUrl: processedFileUrl,
        fileData: processedFile,
        originalFileData: originalFileData,
        trueOriginalFileData: trueOriginalFileData,
        originalImageUrl: originalImageUrl,
        trueOriginalImageUrl: trueOriginalImageUrl,
        thumbnailUrl: thumbnailUrl,
      } : c));
    });
  }, [tx, ty, tScale]);
